cmake_minimum_required(VERSION 3.8)
project(multi_observation_semantic_fusion)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common)
include_directories(include ${PCL_INCLUDE_DIRS})
add_definitions(${PCL_DEFINITIONS})
add_library(fusion_core src/semantic_fusion.cpp)
target_include_directories(fusion_core PUBLIC include)
add_executable(multi_observation_semantic_fusion_node src/multi_observation_semantic_fusion_node.cpp)
target_link_libraries(multi_observation_semantic_fusion_node fusion_core ${PCL_LIBRARIES})
ament_target_dependencies(fusion_core rclcpp sensor_msgs std_msgs pcl_conversions)
ament_target_dependencies(multi_observation_semantic_fusion_node rclcpp sensor_msgs std_msgs pcl_conversions)
install(TARGETS fusion_core multi_observation_semantic_fusion_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch rviz docs DESTINATION share/${PROJECT_NAME})
ament_package()
